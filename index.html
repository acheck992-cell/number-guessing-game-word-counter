import random from collections import Counter from pathlib import Path import string def number_guessing_game(): print("\n=== Number Guessing Game ===") print("Guess a number between 1 and 100!") secret = random.randint(1, 100) attempts = 0 while True: user_input = input("Enter your guess (or q to quit): ").strip() if user_input.lower() == "q": print(f"The secret number was {secret}.") return try: guess = int(user_input) except ValueError: print("Invalid input! Enter a whole number.") continue if not 1 <= guess <= 100: print("Please enter a number from 1 to 100.") continue attempts += 1 if guess < secret: print("Too low! Try a higher number.") elif guess > secret: print("Too high! Try a lower number.") else: score = max(0, 100 - (attempts - 1) * 10) print(f"\nCorrect! You won in {attempts} attempts.") print(f"Your score: {score}/100") return def word_counter(): print("\n=== Text File Word Counter ===") filename = input( "Enter text filename (press Enter for sample.txt): " ).strip() if not filename: filename = "sample.txt" path = Path(filename) if not path.exists(): path.write_text( "Python is easy. Python is useful. " "Practice Python every day.", encoding="utf-8" ) print("Sample text file created.") try: text = Path(filename).read_text(encoding="utf-8") except (OSError, UnicodeError) as error: print(f"Error reading file: {error}") return words = [ word.strip(string.punctuation).lower() for word in text.split() ] words = [word for word in words if word] frequency = Counter(words) print("\n--- Analysis Results ---") print(f"Total words: {len(words)}") print(f"Unique words: {len(frequency)}") print("\nWord Frequency:") if not frequency: print("The file contains no words.") else: for word, count in frequency.most_common(10): print(f"{word}: {count}") def main(): while True: print("\n==============================") print(" Number Guessing & Word Counter") print("==============================") print("1. Number Guessing Game") print("2. Word Counter") print("3. Exit") choice = input("Choose an option (1-3): ").strip() if choice == "1": number_guessing_game() elif choice == "2": word_counter() elif choice == "3": print("Thank you for using the program!") break else: print("Invalid choice! Select 1, 2, or 3.") if __name__ == "__main__": main()
